# OpenAPI reference

This page renders the live OpenAPI schema exported from the running FastAPI
application, using [Swagger UI](https://github.com/swagger-api/swagger-ui).

The schema file (`docs/openapi.json`) is generated by
`scripts/export_openapi.py` and is regenerated automatically as part of the
docs build (see `mkdocs.yml`'s `hooks` and the `docs.yml` workflow). To
refresh it locally:

```bash
uv run python scripts/export_openapi.py
uv run mkdocs serve
```

<script src="https://unpkg.com/swagger-ui-dist@5/swagger-ui-bundle.js"></script>
<script>
  (function () {
    // Swagger UI ships its own dark theme, gated behind an `html.dark-mode`
    // class. Keep it in sync with Material's `data-md-color-scheme` so the
    // Swagger UI panel (including the Schemas section) stays legible when
    // the docs theme is switched to dark mode.
    function syncSwaggerDarkMode() {
      const scheme = document.body.getAttribute("data-md-color-scheme");
      document.documentElement.classList.toggle("dark-mode", scheme === "slate");
    }

    syncSwaggerDarkMode();
    new MutationObserver(syncSwaggerDarkMode).observe(document.body, {
      attributes: true,
      attributeFilter: ["data-md-color-scheme"],
    });
  })();

  window.addEventListener("DOMContentLoaded", function () {
    const specUrl = new URL("../openapi.json", window.location.href).href;
    SwaggerUIBundle({
      url: specUrl,
      dom_id: "#swagger-ui",
      presets: [SwaggerUIBundle.presets.apis],
      layout: "BaseLayout",
    });
  });
</script>
